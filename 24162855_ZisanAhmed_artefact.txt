# First, get borough information for each LSOA

    lsoa_to_borough,
    on='LSOA11',
    how='left'
)

print(f"LSOAs with missing borough codes: {missing_borough}")

# Remove rows with missing borough (they're not in London)
lsoa_with_borough = lsoa_with_borough[lsoa_with_borough['LocalAuthorityCode'].notna()]

# Aggregate to borough level
borough_data = lsoa_with_borough.groupby('LocalAuthorityCode').agg({
    'takeaway_count': 'sum',
    'IMD_Score': 'mean'
}).reset_index()

    'Area Code': 'LocalAuthorityCode',

    'Population': 'population'
})

# Merge with borough data
borough_with_pop = borough_data.merge(
    population_london_clean[['LocalAuthorityCode', 'BoroughName', 'population']],

) * 1000

print("Takeaway density statistics:")

# Show the highest and lowest density boroughs
print("\nTop 5 boroughs by takeaway density:")

print("\nBottom 5 boroughs by takeaway density:")

print("\n✅ Takeaway density calculated successfully!")








print("="*60)

# Get the latest obesity data for each borough (most recent year)

import statsmodels.api as sm
import statsmodels.formula.api as smf
from statsmodels.stats.outliers_influence import variance_inflation_factor
from statsmodels.stats.diagnostic import het_breuschpagan
from statsmodels.stats.stattools import durbin_watson
from scipy import stats
import warnings
warnings.filterwarnings('ignore')

print("="*60)
print("MODEL A: BIVARIATE REGRESSION")
print("Takeaway Density → Obesity Rate")
print("="*60)

# Prepare data
X = final_data['takeaway_density_per_1000']
y = final_data['obesity_rate']

# Add constant (intercept)
X_with_const = sm.add_constant(X)

# Fit the model
model_a = sm.OLS(y, X_with_const).fit()

# Print summary
print(model_a.summary())

# Extract key metrics
print("\n" + "="*60)
print("MODEL A - KEY METRICS")
print("="*60)
print(f"R-squared: {model_a.rsquared:.4f}")
print(f"Adjusted R-squared: {model_a.rsquared_adj:.4f}")
print(f"F-statistic: {model_a.fvalue:.4f}")
print(f"F-statistic p-value: {model_a.f_pvalue:.4f}")
print(f"Number of observations: {int(model_a.nobs)}")
print(f"AIC: {model_a.aic:.2f}")
print(f"BIC: {model_a.bic:.2f}")

print("\n" + "-"*60)
print("COEFFICIENTS:")
print("-"*60)
print(f"Intercept (β0): {model_a.params[0]:.4f}")
print(f"Takeaway Density (β1): {model_a.params[1]:.4f}")
print(f"p-value for β1: {model_a.pvalues[1]:.4f}")

if model_a.pvalues[1] < 0.05:
    print("✅ The coefficient is statistically significant (p < 0.05).")
else:
    print("⚠️ The coefficient is NOT statistically significant (p >= 0.05).")
